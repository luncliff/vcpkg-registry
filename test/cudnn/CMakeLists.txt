cmake_minimum_required(VERSION 3.20)
project(cudnn-discovery NONE)

if(NOT DEFINED TEST_LAYOUT)
  enable_testing()
  foreach(layout IN ITEMS flat archive cuda linux 12.0 12.9 13.0 13.1 13.2 13.3 13.4 old-version)
    add_test(NAME "cudnn-${layout}"
      COMMAND "${CMAKE_COMMAND}" -S "${CMAKE_CURRENT_SOURCE_DIR}"
        -B "${CMAKE_CURRENT_BINARY_DIR}/${layout}" "-DTEST_LAYOUT=${layout}")
  endforeach()
  set_tests_properties(cudnn-old-version PROPERTIES WILL_FAIL TRUE)
  return()
endif()

set(root "${CMAKE_CURRENT_BINARY_DIR}/fixture")
set(include_dir "${root}/include")
set(library_dir "${root}/lib")
set(binary_dir "${root}/bin")
set(version_minor 27)
if(TEST_LAYOUT MATCHES "^[0-9]+\\.[0-9]+$")
  string(APPEND include_dir "/${TEST_LAYOUT}")
  string(APPEND library_dir "/${TEST_LAYOUT}/x64")
  string(APPEND binary_dir "/${TEST_LAYOUT}")
elseif(TEST_LAYOUT STREQUAL "archive")
  string(APPEND library_dir "/x64")
elseif(TEST_LAYOUT STREQUAL "cuda")
  set(include_dir "${root}/cuda/include")
  set(library_dir "${root}/cuda/lib64")
elseif(TEST_LAYOUT STREQUAL "linux")
  set(library_dir "${root}/lib64")
elseif(TEST_LAYOUT STREQUAL "old-version")
  set(version_minor 11)
endif()

file(MAKE_DIRECTORY "${include_dir}" "${library_dir}" "${binary_dir}")
file(WRITE "${include_dir}/cudnn.h" "#include \"cudnn_version.h\"\n")
file(WRITE "${include_dir}/cudnn_version.h"
  "#define CUDNN_MAJOR 9\n#define CUDNN_MINOR ${version_minor}\n#define CUDNN_PATCHLEVEL 0\n")
if(TEST_LAYOUT STREQUAL "linux" OR TEST_LAYOUT STREQUAL "cuda")
  set(library "${library_dir}/libcudnn.so")
else()
  set(CMAKE_FIND_LIBRARY_PREFIXES "")
  set(CMAKE_FIND_LIBRARY_SUFFIXES ".lib")
  set(library "${library_dir}/cudnn.lib")
  file(WRITE "${binary_dir}/cudnn64_9.dll" "")
endif()
file(WRITE "${library}" "")

set(ENV{CUDNN_ROOT_DIR} "${root}")
set(ENV{CUDA_PATH} "")
set(ENV{CUDA_TOOLKIT_ROOT_DIR} "")
set(ENV{cudnn} "")
set(ENV{CUDNN} "")
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../ports/cudnn")
find_package(CUDNN 9.27.0 REQUIRED)

if(NOT CUDNN_VERSION STREQUAL "9.27.0" OR NOT CUDNN_VERSION_MAJOR STREQUAL "9"
    OR NOT CUDNN_INCLUDE_DIRS STREQUAL include_dir OR NOT CUDNN_LIBRARIES STREQUAL library)
  message(FATAL_ERROR "Incorrect cuDNN version or paths")
endif()
get_target_property(location CuDNN::CuDNN IMPORTED_LOCATION)
get_target_property(target_type CuDNN::CuDNN TYPE)
if(TEST_LAYOUT STREQUAL "linux" OR TEST_LAYOUT STREQUAL "cuda")
  if(NOT target_type STREQUAL "UNKNOWN_LIBRARY" OR NOT location STREQUAL library)
    message(FATAL_ERROR "Incorrect cuDNN library target")
  endif()
else()
  get_target_property(implib CuDNN::CuDNN IMPORTED_IMPLIB)
  if(NOT target_type STREQUAL "SHARED_LIBRARY" OR NOT implib STREQUAL library
      OR NOT location STREQUAL "${binary_dir}/cudnn64_9.dll")
    message(FATAL_ERROR "Incorrect cuDNN DLL target")
  endif()
endif()
